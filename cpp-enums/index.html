02. The Investigation [ASM]

ANALYSIS: ASSEMBLY, PIPELINE, & STACK SPILLS

[A] THE CMOV vs BRANCH AXIOM

Axiom: x86-64 has `cmov` for 32-bit but NO conditional move for 8-bit.

INT (32-bit)cmovnel %ecx, %eax
CHAR (8-bit)jne .LBB4_6

[B] RAW ASSEMBLY COMPARISON

INT TERNARY (FAST PATH)
movl -12(%rbp), %edx   # Load i
andl $1, %edx          # i & 1
xorl %eax, %eax        # 0
movl $1, %ecx          # 1
cmpl $0, %edx          # Compare
cmovnel %ecx, %eax     # CMOV (No Branch)
CHAR TERNARY (SLOW PATH)
movl -12(%rbp), %ecx   # Load i
andl $1, %ecx          # i & 1
movb $83, %al          # 'S'
movb $66, %dl          # 'B'
movb %dl, -42(%rbp)    # SPILL STACK
cmpl $0, %ecx          # Compare
movb %al, -41(%rbp)    # SPILL STACK
jne .LBB4_6            # BRANCH
movb -42(%rbp), %al    # RELOAD STACK
